namespace ConsoleApp3
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int num = 20;
            string c = "a";
            string _tmp1 = "";
            string _tmp2 = "";
            string _txt = "";

            for (int i = 0; i < num; i++)
            {
                _tmp1 = "";
                _tmp2 = "";

                for (int j = 0; j < num-i; j++)
                {
                    _tmp1 += " ";
                }

                for (int j = 0; j < 2*i-1; j++)
                {
                    _tmp2 += c;
                }

                for (int j = 0; j < 3; j++)
                {
                    _txt += _tmp1+_tmp2;
                    for (int k = 0; k < num-i; k++)
                    {
                        _txt += " ";
                    }
                }
                _txt += "\n";

            }
            Console.WriteLine(_txt);
        }
    }
}
